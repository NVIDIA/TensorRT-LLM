file(GLOB_RECURSE SRC_CPP *.cpp)
file(GLOB_RECURSE SRC_CU *.cu)

add_library(trtllm_gen_batched_gemm OBJECT)
filter_source_cuda_architectures(SOURCE_LIST SRC_CPP ARCHS 100 103 107 100f)

target_sources(trtllm_gen_batched_gemm PRIVATE ${SRC_CPP} ${SRC_CU})
target_compile_definitions(trtllm_gen_batched_gemm
                           PUBLIC TLLM_GEN_EXPORT_INTERFACE TLLM_ENABLE_CUDA)

set_property(TARGET trtllm_gen_batched_gemm PROPERTY POSITION_INDEPENDENT_CODE
                                                     ON)
set_property(TARGET trtllm_gen_batched_gemm PROPERTY CUDA_RESOLVE_DEVICE_SYMBOLS
                                                     ON)

# With no SM10x architecture enabled, every entry of the generated kernel list
# in trtllmGen_bmm_export/KernelMetaInfo.h is compiled out and the list is an
# empty array. GCC then reports -Warray-bounds for indexing it in code that such
# a build can never reach, so silence that warning for this configuration only.
get_directory_property(trtllm_gen_defs COMPILE_DEFINITIONS)
if("EXCLUDE_SM_100" IN_LIST trtllm_gen_defs
   AND "EXCLUDE_SM_100F" IN_LIST trtllm_gen_defs
   AND "EXCLUDE_SM_103" IN_LIST trtllm_gen_defs
   AND "EXCLUDE_SM_107" IN_LIST trtllm_gen_defs)
  target_compile_options(trtllm_gen_batched_gemm
                         PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wno-array-bounds>)
endif()

# trtllmGen_bmm_export/ is generated by trtllm-gen and synced into this repo;
# its BatchedGemmInterface.h stores a few private members (mRankId,
# mExportsCubin, mNumRotations) that the exported code never reads, which Clang
# reports. Fix that in the generator rather than here.
target_compile_options(
  trtllm_gen_batched_gemm
  PRIVATE
    $<$<AND:$<COMPILE_LANGUAGE:CXX>,$<CXX_COMPILER_ID:Clang>>:-Wno-unused-private-field>
)

# Embed cubins via INCBIN.
include(tllm_cubin_archive)
tllm_add_cubin_archive_sources(
  trtllm_gen_batched_gemm
  "${CMAKE_CURRENT_SOURCE_DIR}/trtllmGen_bmm_export/cubins"
  NAMESPACE batchedGemm tensorrt_llm ${TRTLLM_ABI_NAMESPACE} kernels
  ARCHS 100 103 107 100f)
